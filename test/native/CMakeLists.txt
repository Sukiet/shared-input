project(shared_input_safety_tests LANGUAGES CXX)
enable_testing()

target_compile_features(safety_policy_test PRIVATE cxx_std_17)
add_test(NAME safety_policy COMMAND safety_policy_test)

  target_link_libraries(keyboard_capture_test PRIVATE user32)
  add_test(NAME keyboard_capture COMMAND keyboard_capture_test)
